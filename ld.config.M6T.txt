# Bionic loader config for the Meizu M6T, based on
# system/core/rootdir/etc/ld.config.legacy.txt.
#
# The Nougat fingerprint.default.so, loaded by the fingerprint HAL service in
# the [vendor] namespace, needs the Nougat libkeymaster_messages.so and
# libkeymaster1.so; the system copies have a different C++ mangling. Search
# /vendor/${LIB}/fp_a7, which holds the Nougat copies, before /system/${LIB}.
# The [legacy] namespace (keystore and other /system processes) is unchanged;
# goodixfingerprintd gets fp_a7 through LD_LIBRARY_PATH in microtrust.rc.

# Processes from /vendor/bin/ run in the [vendor] namespace.
# Processes from /system/ or /sbin/ run in the [legacy] namespace.
dir.vendor = /vendor/bin/
dir.vendor = /system/vendor/bin/
dir.legacy = /system
dir.legacy = /vendor
dir.legacy = /sbin

[vendor]
# fp_a7 before /system/${LIB}; otherwise identical to the legacy config.
namespace.default.isolated = false
namespace.default.search.paths = /vendor/${LIB}/hw:/vendor/${LIB}/egl:/vendor/${LIB}:/vendor/${LIB}/fp_a7:/system/${LIB}
namespace.default.asan.search.paths = /data/asan/vendor/${LIB}/hw:/vendor/${LIB}/hw:/data/asan/vendor/${LIB}/egl:/vendor/${LIB}/egl:/data/asan/vendor/${LIB}:/vendor/${LIB}:/vendor/${LIB}/fp_a7:/data/asan/system/${LIB}:/system/${LIB}

[legacy]
# Unchanged from ld.config.legacy.txt.
namespace.default.isolated = false
namespace.default.search.paths = /system/${LIB}:/vendor/${LIB}
namespace.default.asan.search.paths = /data/asan/system/${LIB}:/system/${LIB}:/data/asan/vendor/${LIB}:/vendor/${LIB}
